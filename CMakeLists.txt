cmake_minimum_required(VERSION 3.24)
project(ovr_skins LANGUAGES C CXX)
include(FetchContent)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
FetchContent_Declare(protobuf
  URL https://github.com/protocolbuffers/protobuf/archive/refs/tags/v21.12.tar.gz
  URL_HASH SHA256=22fdaf641b31655d4b2297f9981fa5203b2866f8332d3c6333f6b0107bb320de
  SOURCE_SUBDIR cmake)
FetchContent_MakeAvailable(protobuf)
file(GLOB OVR_PROTOS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build/protos/*.proto")
list(APPEND OVR_PROTOS "${CMAKE_CURRENT_SOURCE_DIR}/proto/ovr_plus.proto")
if(NOT OVR_PROTOS)
  message(FATAL_ERROR "Run scripts/fetch-protos.ps1 first")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(OVR_GENERATED)
foreach(PROTO ${OVR_PROTOS})
  get_filename_component(NAME ${PROTO} NAME_WLE)
  set(CC "${CMAKE_CURRENT_BINARY_DIR}/generated/${NAME}.pb.cc")
  set(H "${CMAKE_CURRENT_BINARY_DIR}/generated/${NAME}.pb.h")
  add_custom_command(OUTPUT ${CC} ${H}
    COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR}/generated
      -I ${CMAKE_CURRENT_SOURCE_DIR}/build/protos -I ${CMAKE_CURRENT_SOURCE_DIR}/proto -I ${protobuf_SOURCE_DIR}/src ${PROTO}
    DEPENDS protobuf::protoc ${OVR_PROTOS} VERBATIM)
  list(APPEND OVR_GENERATED ${CC} ${H})
endforeach()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
file(GLOB_RECURSE OVR_SOURCES CONFIGURE_DEPENDS upstream/src/*.cpp)
add_library(ovr_protocol STATIC ${OVR_GENERATED})
target_include_directories(ovr_protocol PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ovr_protocol PUBLIC protobuf::libprotobuf)
target_compile_options(ovr_protocol PRIVATE /utf-8 /MP6)
set_property(TARGET ovr_protocol PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ovr_skins SHARED ${OVR_SOURCES}
  upstream/src/Resource/Minhook/src/buffer.c
  upstream/src/Resource/Minhook/src/hook.c
  upstream/src/Resource/Minhook/src/trampoline.c
  upstream/src/Resource/Minhook/src/hde/hde64.c)
target_include_directories(ovr_skins PRIVATE upstream/src upstream/src/Resource/Minhook/include)
target_include_directories(ovr_skins PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ovr_skins PRIVATE ovr_protocol winhttp)
target_compile_features(ovr_skins PRIVATE cxx_std_20)
target_compile_definitions(ovr_skins PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(ovr_skins PRIVATE /utf-8 /W3 "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
set_property(TARGET ovr_skins PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Every configure links to a fresh directory: a development game can keep using
# the previous DLL while its successor is being built.
string(RANDOM LENGTH 12 ALPHABET 0123456789abcdef OVR_BUILD_ID)
set(OVR_DLL_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/package/${OVR_BUILD_ID}/Release")
set_target_properties(ovr_skins PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${OVR_DLL_DIRECTORY}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ovr-dll-path.txt" CONTENT "${OVR_DLL_DIRECTORY}/ovr_skins.dll\n")
add_custom_command(TARGET ovr_skins POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources/assistant" "$<TARGET_FILE_DIR:ovr_skins>/assistant"
  COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/stage-config.ps1"
    -Workspace "${CMAKE_CURRENT_SOURCE_DIR}" -Destination "$<TARGET_FILE_DIR:ovr_skins>"
  VERBATIM)
